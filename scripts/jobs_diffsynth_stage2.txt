# Diffsynth, stage 2: the resumes. Run only after stage 1 has finished.
#
#   python scripts/gpu_queue.py --jobs scripts/jobs_diffsynth_stage2.txt
#
# Each line resumes the matching stage-1 pretrain and runs it to epoch 400.
# ds_run.sh checks the checkpoint exists and fails loudly if not, so launching
# this early costs a fast error rather than a wasted card.
#
# resume_synth continues in-domain (paper's "Synth"); resume_real switches the
# training loader to NSynth (paper's "Real"). Both keep validating on both
# domains, which is where val_id/* and val_ood/* come from.
#
# The two resumes from one pretrain are independent, so they can occupy two
# cards at once -- that is the parallelism this stage has, and why it is worth
# a separate queue rather than chaining each variant on one GPU.
RESUME=pre_hybrid    scripts/ds_run.sh synth_hybrid    resume_synth "{gpu}"
RESUME=pre_hybrid    scripts/ds_run.sh real_hybrid     resume_real  "{gpu}"
RESUME=pre_mag       scripts/ds_run.sh synth_mag       resume_synth "{gpu}" model.sw_loss.log_mag_w=0
RESUME=pre_mag       scripts/ds_run.sh real_mag        resume_real  "{gpu}" model.sw_loss.log_mag_w=0
RESUME=pre_log       scripts/ds_run.sh synth_log       resume_synth "{gpu}" model.sw_loss.mag_w=0
RESUME=pre_log       scripts/ds_run.sh real_log        resume_real  "{gpu}" model.sw_loss.mag_w=0
RESUME=pre_mag_halfw scripts/ds_run.sh synth_mag_halfw resume_synth "{gpu}" model.sw_loss.log_mag_w=0 schedule.sw_w.end_v=0.5
RESUME=pre_mag_halfw scripts/ds_run.sh real_mag_halfw  resume_real  "{gpu}" model.sw_loss.log_mag_w=0 schedule.sw_w.end_v=0.5
