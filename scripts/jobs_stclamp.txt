# stclamp ladder: one arm per GPU, picked up as cards free.
# NORM=batch is not optional -- the tanh ladder these are compared against
# ran --norm batch, and eps_ladder.sh defaults NORM to group. Leaving it out
# changes two things at once and makes the run unattributable.
# Every other setting is eps_ladder.sh's default, which is the tanh ladder's.
NORM=batch HEAD_BOUND=stclamp OUT=results/ddsp/eps_ladder_stclamp ARMS="L1_STFT" scripts/eps_ladder.sh "{gpu}"
